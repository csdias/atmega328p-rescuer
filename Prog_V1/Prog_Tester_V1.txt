#include <SPI.h>

/*
 * GESTÃO DE BARRAMENTO - ARDUINO MEGA (MASTER)
 */

const int PIN_CONTROL_USBASP = 49; 
const int PIN_CONTROL_MASTER = 48; 

const int MEGA_MISO = 50;
const int MEGA_MOSI = 51;
const int MEGA_SCK  = 52;
const int MEGA_SS   = 53;

void setup() {
  pinMode(PIN_CONTROL_USBASP, OUTPUT);
  pinMode(PIN_CONTROL_MASTER, OUTPUT);

  cancelarAcesso();

  Serial.begin(9600);   
  Serial2.begin(9600);  
  
  SPI.begin();
  // Abrandar o SPI ajuda na estabilidade com cabos comuns
  SPI.setClockDivider(SPI_CLOCK_DIV16); 
  
  while (!Serial); 
  exibirMenu();
}

void loop() {
  if (Serial.available() > 0) {
    String comando = Serial.readStringUntil('\n');
    comando.trim(); 

    if (comando == "USBASP_SPI") { ativarUSBasp(); } 
    else if (comando == "uC_MASTER_SPI") { ativarMegaMaster(); } 
    else if (comando == "SPI_CANCEL_ACCESS") { cancelarAcesso(); Serial.println("\n[SISTEMA] Isolado."); }
    else if (comando == "TEST_SERIAL2_COM") { testarConexaoNano(); }
    else if (comando == "SPI_TEST") { executarSpiTest(); }
    else { Serial.println("\n[ERRO] Comando invalido!"); }
    exibirMenu();
  }
}

void executarSpiTest() {
  Serial.println("\n[SPI TEST] Iniciando sequencia...");
  
  // Limpeza de segurança
  while(Serial2.available()) Serial2.read(); 
  delay(100); 
  
  // 1. Aviso via Serial
  Serial2.println("SPI_TEST");
  
  unsigned long timeout = millis();
  bool pronto = false;
  while (millis() - timeout < 3000) {
    if (Serial2.available() > 0) {
      String resp = Serial2.readStringUntil('\n');
      resp.trim();
      if (resp == "COM_SET") { pronto = true; break; }
    }
  }

  if (!pronto) {
    Serial.println("[ERRO] Nano nao confirmou COM_SET. Tente novamente.");
    return;
  }

  // 2. Ativação física
  ativarMegaMaster(); 
  delay(200); // Tempo para o Nano entrar no modo Slave ISR

  // 3. Transação SPI
  digitalWrite(MEGA_SS, LOW); 
  delay(10); 
  
  // Enviar comando "LISTEN\n"
  const char* msg = "LISTEN\n";
  for (int i = 0; i < strlen(msg); i++) {
    SPI.transfer(msg[i]);
    delay(2);
  }
  
  // 4. Puxar resposta (Dummy Bytes)
  // Aumentado para 40 ciclos para garantir o fim da string
  String respostaSpi = "";
  for (int i = 0; i < 40; i++) {
    char c = SPI.transfer(0x00); 
    if (c != 0 && (byte)c != 255 && c != '\r') {
      respostaSpi += c;
    }
    delay(5);
  }

  digitalWrite(MEGA_SS, HIGH); 
  
  // 5. Validação
  if (respostaSpi.indexOf("NANO_SPI_OK") >= 0) {
    Serial.println("\n*******************************");
    Serial.println("*  COMUNICACAO BEM SUCEDIDA!  *");
    Serial.println("*******************************");
  } else {
    Serial.print("[SPI RESULTADO INCOMPLETO]: ");
    Serial.println(respostaSpi);
  }
}

void cancelarAcesso() {
  digitalWrite(PIN_CONTROL_USBASP, LOW);
  digitalWrite(PIN_CONTROL_MASTER, LOW);
  desativarPinosSPIMaster();
}

void ativarUSBasp() {
  digitalWrite(PIN_CONTROL_MASTER, LOW);
  desativarPinosSPIMaster();
  delay(50);
  digitalWrite(PIN_CONTROL_USBASP, HIGH);
  Serial.println("\n[MODO] USBasp Ativo.");
}

void ativarMegaMaster() {
  digitalWrite(PIN_CONTROL_USBASP, LOW);
  delay(50);
  ativarPinosSPIMaster();
  digitalWrite(PIN_CONTROL_MASTER, HIGH);
}

void desativarPinosSPIMaster() {
  pinMode(MEGA_MOSI, INPUT);
  pinMode(MEGA_MISO, INPUT);
  pinMode(MEGA_SCK, INPUT);
  pinMode(MEGA_SS, INPUT); 
}

void ativarPinosSPIMaster() {
  pinMode(MEGA_MOSI, OUTPUT);
  pinMode(MEGA_SCK, OUTPUT);
  pinMode(MEGA_SS, OUTPUT);
  pinMode(MEGA_MISO, INPUT);
}

void testarConexaoNano() {
  while(Serial2.available()) Serial2.read(); 
  Serial2.println("LISTEN");
  unsigned long t = millis();
  while (millis() - t < 1000) {
    if (Serial2.available() > 0) {
      Serial.println("[NANO]: " + Serial2.readStringUntil('\n'));
      return;
    }
  }
  Serial.println("[ERRO] Serial 2 offline.");
}

void exibirMenu() {
  Serial.println("\n--- MENU ---");
  Serial.println(" USBASP_SPI / uC_MASTER_SPI / SPI_CANCEL_ACCESS");
  Serial.println(" TEST_SERIAL2_COM / SPI_TEST");
  Serial.print("Comando: ");
}
